function openCropper(input) { if (input.files && input.files[0]) { const reader = new FileReader(); reader.onload = function(e) { // 先顯示視窗，再初始化 Croppie document.getElementById('cropperModal').style.display = 'block'; const el = document.getElementById('cropper-instance'); // 確保每次開啟都是新的實例 if (croppieInstance) { croppieInstance.destroy(); } croppieInstance = new Croppie(el, { viewport: { width: 280, height: 186, type: 'square' }, // 調整為適合手機的寬度 boundary: { width: '100%', height: 300 }, showZoomer: true, enableOrientation: true, mouseWheelZoom: 'ctrl' // 避免滾輪誤觸縮放 }); // 綁定圖片 croppieInstance.bind({ url: e.target.result }).then(() => { console.log('圖片載入成功'); }); }; reader.readAsDataURL(input.files[0]); } }
